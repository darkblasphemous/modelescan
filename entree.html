function afficherResultat(titre, nom, idMembre, heureTexte, type, photoUrl) { const msgBox = document.getElementById("statusMsg"); const photoContainer = document.getElementById("photoContainer"); const userPhoto = document.getElementById("userPhoto"); // Sécurité : si l'élément n'existe pas encore dans la page, on arrête pour éviter le bug if (!msgBox) return; msgBox.style.display = "block"; msgBox.className = "status-box"; if (type === "success") { msgBox.classList.add("status-success"); } else if (type === "warning") { msgBox.classList.add("status-warning"); } else if (type === "error") { msgBox.classList.add("status-error"); } else { msgBox.classList.add("status-loading"); } if (type === "loading") { msgBox.innerText = titre; } else { let heureHtml = heureTexte ? `
Heure : ${heureTexte}
` : ''; msgBox.innerHTML = `
${titre}
${nom}
${heureHtml} `; } if (photoContainer && userPhoto) { if (photoUrl && photoUrl.trim() !== "" && photoUrl.length > 5) { userPhoto.src = photoUrl; photoContainer.style.display = "block"; } else { photoContainer.style.display = "none"; } } }
