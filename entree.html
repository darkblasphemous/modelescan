// ---------------------------------------------------------------------- // FONCTION : Validation du scan d'entrée // ---------------------------------------------------------------------- function validerEntree(idScanne) { // Vérification de sécurité locale if (!idScanne || idScanne.trim() === "") { showErrorScreen("Code QR ou ID invalide."); return; } // Afficher l'écran de chargement showLoadingScreen(); // Nettoyage des anciens scripts JSONP const oldScript = document.getElementById('jsonp-scan-script'); if (oldScript) oldScript.remove(); // Construction propre de l'URL avec action=entree const cleanId = idScanne.trim(); const url = `${SCRIPT_URL}?action=entree&id=${encodeURIComponent(cleanId)}&callback=handleScanResponse&_=${new Date().getTime()}`; // Création et injection du script const script = document.createElement('script'); script.id = 'jsonp-scan-script'; script.src = url; script.onerror = function() { showErrorScreen("Erreur de connexion au serveur."); }; document.body.appendChild(script); } // ---------------------------------------------------------------------- // FONCTION : Réception de la réponse du serveur // ---------------------------------------------------------------------- window.handleScanResponse = function(response) { console.log("Réponse serveur :", response); if (response.status === "success") { // Entrée validée avec succès ! let photo = convertDriveUrl(response.photoUrl); showSuccessScreen(response.nom, photo, response.heure); } else if (response.status === "already_in") { // Déjà enregistré comme présent let photo = convertDriveUrl(response.photoUrl); showAlreadyInScreen(response.nom, photo, response.heure); } else { // Erreur (ex: ID non trouvé) showErrorScreen(response.message || "Erreur lors de la validation."); } };
