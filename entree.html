// ---------------------------------------------------------------------- // FONCTION : Validation du scan d'entrée // ---------------------------------------------------------------------- function validerEntree(idScanne) { // Afficher l'écran de chargement / validation showLoadingScreen(); // Nettoyage des anciens scripts JSONP const oldScript = document.getElementById('jsonp-scan-script'); if (oldScript) oldScript.remove(); // URL complète avec action=scanIn ET id=... const url = `${SCRIPT_URL}?action=scanIn&id=${encodeURIComponent(idScanne)}&callback=handleScanResponse&_=${new Date().getTime()}`; // Création et injection de la balise script const script = document.createElement('script'); script.id = 'jsonp-scan-script'; script.src = url; // Gestion d'erreur réseau / timeout script.onerror = function() { showErrorScreen("Erreur de connexion au serveur."); }; document.body.appendChild(script); } // ---------------------------------------------------------------------- // FONCTION : Réception de la réponse du serveur // ---------------------------------------------------------------------- window.handleScanResponse = function(response) { console.log("Réponse serveur :", response); if (response.status === "success") { // Entrée validée avec succès ! let photo = convertDriveUrl(response.photoUrl); showSuccessScreen(response.nom, photo, response.heure); } else if (response.status === "already_in") { // Déjà enregistré comme présent let photo = convertDriveUrl(response.photoUrl); showAlreadyInScreen(response.nom, photo, response.heure); } else { // Erreur (ex: ID non trouvé) showErrorScreen(response.message || "Erreur lors de la validation."); } };
